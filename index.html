import time from itertools import combinations # 1. 사용자 입력 데이터 설정 BUDGET = 250000 # 총 예산 TARGET_COUNT = 3 # 관극 목표 횟수 # 배우별 선호도 (1~10점) actor_prefs = { "A": 10, "B": 9, "C": 7, "D": 5 } # 배우 조합 가산점 (-5 ~ +5점) pair_bonuses = { tuple(sorted(["A", "B"])): 5, tuple(sorted(["A", "C"])): 2, tuple(sorted(["B", "D"])): 3, } # 가상의 공연 회차 데이터 (15회차) shows = [ {"id": 1, "date": "11/01", "cast": ["A", "C"], "price": 80000}, {"id": 2, "date": "11/02", "cast": ["B", "D"], "price": 70000}, {"id": 3, "date": "11/03", "cast": ["A", "B"], "price": 95000}, {"id": 4, "date": "11/04", "cast": ["C", "D"], "price": 75000}, {"id": 5, "date": "11/05", "cast": ["A", "D"], "price": 80000}, {"id": 6, "date": "11/06", "cast": ["B", "C"], "price": 85000}, {"id": 7, "date": "11/07", "cast": ["A", "B"], "price": 90000}, {"id": 8, "date": "11/08", "cast": ["C", "D"], "price": 70000}, {"id": 9, "date": "11/09", "cast": ["A", "C"], "price": 85000}, {"id": 10, "date": "11/10", "cast": ["B", "D"], "price": 75000}, {"id": 11, "date": "11/11", "cast": ["A", "D"], "price": 85000}, {"id": 12, "date": "11/12", "cast": ["B", "C"], "price": 80000}, {"id": 13, "date": "11/13", "cast": ["A", "B"], "price": 100000}, {"id": 14, "date": "11/14", "cast": ["A", "C"], "price": 80000}, {"id": 15, "date": "11/15", "cast": ["B", "D"], "price": 70000}, ] # 회차별 기본 만족도 계산 함수 def calc_show_score(show): base = sum(actor_prefs[actor] for actor in show["cast"]) pair_key = tuple(sorted(show["cast"])) bonus = pair_bonuses.get(pair_key, 0) return base + bonus for s in shows: s["score"] = calc_show_score(s) # ========================================== # 알고리즘 1: 완전탐색 (Brute Force) # ========================================== def solve_brute_force(shows, budget, k): start_time = time.perf_counter() best_score = -1 best_comb = None checked_count = 0 for comb in combinations(shows, k): checked_count += 1 total_price = sum(s["price"] for s in comb) if total_price <= budget: total_score = sum(s["score"] for s in comb) if total_score > best_score: best_score = total_score best_comb = comb elapsed = (time.perf_counter() - start_time) * 1000 return best_score, best_comb, checked_count, elapsed # ========================================== # 알고리즘 2: 그리디 (Greedy - 가성비순 정렬) # ========================================== def solve_greedy(shows, budget, k): start_time = time.perf_counter() # 만족도/가격 비율(가성비)이 높은 순으로 정렬 sorted_shows = sorted(shows, key=lambda x: x["score"] / x["price"], reverse=True) selected = [] current_cost = 0 checked_count = 0 for s in sorted_shows: checked_count += 1 if len(selected) < k and current_cost + s["price"] <= budget: selected.append(s) current_cost += s["price"] total_score = sum(s["score"] for s in selected) if len(selected) == k else -1 elapsed = (time.perf_counter() - start_time) * 1000 return total_score, selected, checked_count, elapsed # ========================================== # 알고리즘 3: 동적 계획법 (DP - 1,000원 단위 압축) # ========================================== def solve_dp(shows, budget, k): start_time = time.perf_counter() unit = 1000 # 메모리/속도 최적화를 위해 1,000원 단위로 스케일링 max_b = budget // unit n = len(shows) # dp[count][b] = (최대 점수, 선택한 회차 ID 리스트) dp = [[(-1, []) for _ in range(max_b + 1)] for _ in range(k + 1)] dp[0][0] = (0, []) ops_count = 0 for idx, s in enumerate(shows): cost = s["price"] // unit score = s["score"] # 역순 순회로 회차 중복 선택 방지 (0-1 Knapsack) for c in range(k, 0, -1): for b in range(max_b, cost - 1, -1): ops_count += 1 prev_score, prev_list = dp[c - 1][b - cost] if prev_score != -1 and prev_score + score > dp[c][b][0]: dp[c][b] = (prev_score + score, prev_list + [s]) best_score = -1 best_comb = [] for b in range(max_b + 1): if dp[k][b][0] > best_score: best_score, best_comb = dp[k][b] elapsed = (time.perf_counter() - start_time) * 1000 return best_score, best_comb, ops_count, elapsed # 결과 출력 및 비교 for name, func in [("완전탐색", solve_brute_force), ("그리디", solve_greedy), ("동적계획법(DP)", solve_dp)]: score, comb, ops, ms = func(shows, BUDGET, TARGET_COUNT) total_cost = sum(s["price"] for s in comb) if comb else 0 dates = [f"{s['date']}({'+'.join(s['cast'])})" for s in comb] if comb else [] print(f"[{name}]") print(f" - 총 만족도: {score}점 | 총 비용: {total_cost:,}원 | 탐색/연산 수: {ops}회 | 소요시간: {ms:.4f}ms") print(f" - 선택 회차: {', '.join(dates)}\n")
